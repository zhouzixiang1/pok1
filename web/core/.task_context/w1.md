## `float_lead.py` — Concrete code skeleton

```python
"""Turn float-lead polarized mechanism (v115 offensive pivot).

Triggers when villain was preflop raiser (PFR), villain c-bet flop, we called,
and villain checked turn. This sequence caps villain's range; we attack with a
polarized lead (value for thin/strong/nut, bluff for gutshot/BDFD/two-overs on
safe runouts). NEW decision point not touched by v74-v112.
"""

# Frequency cap for the BLUFF branch (value branch fires every eligible spot)
_FLOAT_LEAD_BLUFF_FREQ = 0.40

# Maximum board wetness allowed for bluff floats
_FLOAT_LEAD_BLUFF_MAX_WETNESS = 0.45


def _sequence_matches_float_lead(history, my_id, opponent_id):
    """Return True iff history shows: PRE villain raised, FLOP villain bet & we called,
    TURN villain just checked. Caller already confirms last_opp_action_type=='check'."""
    villain_pre_raise = False
    villain_flop_bet = False
    we_called_flop = False
    for r in history:
        rd = r.get('round')
        pid = r.get('player_id')
        at = r.get('action_type')
        if rd == 0 and pid == opponent_id and at in ('raise', 'allin'):
            villain_pre_raise = True
        if rd == 1 and pid == opponent_id and at in ('raise', 'allin'):
            villain_flop_bet = True
        if rd == 1 and pid == my_id and at == 'call' and villain_flop_bet:
            we_called_flop = True
        if rd == 2 and pid == opponent_id and at in ('raise', 'allin'):
            return False  # villain already aggressive on turn — not a check
    return villain_pre_raise and villain_flop_bet and we_called_flop


def should_float_lead_turn(round_idx, to_call, spot_info, history, my_id, opponent_id,
                            my_cards, public_cards, made_strength, draw_info,
                            value_profile, board_texture, opponent_model):
    """Return {'active': bool, 'ratio': float, 'polarity': 'value'|'bluff'}."""
    inactive = {'active': False, 'ratio': 0.0, 'polarity': 'none'}
    if round_idx != 2 or to_call != 0:
        return inactive
    if spot_info.get('last_opp_action_type') != 'check':
        return inactive
    if not _sequence_matches_float_lead(history, my_id, opponent_id):
        return inactive

    tier = value_profile.get('tier', 'none') if value_profile else 'none'
    # ── VALUE LEAD: tier strong/nut OR made_strength >= 0.55 ──
    if tier in ('strong', 'nut') or made_strength >= 0.55:
        ratio = 0.62 + 0.10 * max(0.0, made_strength - 0.55)
        # opponent calling-station amplification
        conf = opponent_model.get('confidence', 0.0) if opponent_model else 0.0
        fold_turn = opponent_model.get('fold_to_bet_turn', 0.44) if opponent_model else 0.44
        if conf >= 0.20 and fold_turn <= 0.30:
            ratio += 0.05 * conf  # sticky villain → size up for thin value
        ratio = min(ratio, 0.85)
        return {'active': True, 'ratio': ratio, 'polarity': 'value'}

    # ── BLUFF LEAD: gutshot/BDFD/two-overs on safe runout ──
    wetness = board_texture.get('wetness', 1.0) if board_texture else 1.0
    if wetness > _FLOAT_LEAD_BLUFF_MAX_WETNESS:
        return inactive
    if made_strength >= 0.25:
        return inactive
    dtype = draw_info.get('type', 'none') if draw_info else 'none'
    dquality = draw_info.get('quality', 0.0) if draw_info else 0.0
    # two overcards check
    if len(public_cards) >= 3 and len(my_cards) >= 2:
        from card_utils import card_number
        board_ranks = [card_number(c) for c in public_cards]
        my_ranks = [card_number(c) for c in my_cards]
        top_board = max(board_ranks) if board_ranks else 0
        two_overs = all(r > top_board for r in my_ranks)
    else:
        two_overs = False
    has_bluff_equity = dtype in ('gutshot', 'backdoor_flush') or dquality >= 0.08 or two_overs
    if not has_bluff_equity:
        return inactive
    # frequency cap via hand+board hash
    roll = (hash(tuple(my_cards) + tuple(public_cards)) % 1000) / 1000.0
    if roll >= _FLOAT_LEAD_BLUFF_FREQ:
        return inactive
    # only bluff vs opponents likely to fold the turn
    conf = opponent_model.get('confidence', 0.0) if opponent_model else 0.0
    fold_turn = opponent_model.get('fold_to_bet_turn', 0.44) if opponent_model else 0.44
    if conf >= 0.20 and fold_turn < 0.40:
        return inactive
    ratio = 0.62
    return {'active': True, 'ratio': ratio, 'polarity': 'bluff'}


def float_lead_sizing(ratio, to_call, pot, min_raise, my_chips, my_round_bet):
    pot_after_call = pot + to_call
    target = int(to_call + pot_after_call * ratio)
    amount = max(min_raise, target)
    amount = min(amount, my_chips - 1)
    if amount <= to_call or amount < min_raise:
        return None
    return amount
```
