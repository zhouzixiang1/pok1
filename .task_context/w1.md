```python
def _count_opp_check_raises(hand_requests, my_id, opponent_id):
    """Aggregate opponent check-then-raise events across prior hands.

    A check-raise event = opponent performed >=1 check AND >=1 raise/allin
    in the SAME postflop street (round 1/2/3) of a single hand. Returns
    (events, opportunities) where opportunities = number of hands where the
    opponent played any postflop street. Used to smooth a per-opponent
    `check_raise_freq` signal that distinguishes habitual check-raisers.
    """
    events = 0
    opportunities = 0
    for req in hand_requests:
        if opponent_can_lock_win(req, my_id):
            continue
        history = req.get("history", [])
        if not history:
            continue
        per_round_check = {}
        per_round_raise = {}
        for rec in history:
            if rec.get("player_id") != opponent_id:
                continue
            ridx = rec.get("round", 0)
            if ridx == 0:
                continue
            atype = rec.get("action_type")
            if atype == "check":
                per_round_check[ridx] = per_round_check.get(ridx, 0) + 1
            elif atype in ("raise", "allin"):
                per_round_raise[ridx] = per_round_raise.get(ridx, 0) + 1
        any_postflop_street = bool(per_round_check or per_round_raise)
        if not any_postflop_street:
            continue
        opportunities += 1
        for ridx in set(list(per_round_check.keys()) + list(per_round_raise.keys())):
            if per_round_check.get(ridx, 0) >= 1 and per_round_raise.get(ridx, 0) >= 1:
                events += 1
                break
    return events, opportunities
```
