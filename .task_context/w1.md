## Code skeleton (use this exact signature)

```python
def _flop_thin_value_cbet_eligible(round_idx, to_call, spot_info, made_strength,
                                    value_profile, board_texture, opponent_model):
    """Thin-value flop cbet trigger: marginal made hands on DRY boards IN POSITION
    vs opponents with fold_to_bet_flop 0.45-0.60. Fills the gap where existing
    flop cbet logic (medium=0.50 win_rate OR made_strength>=0.62 OR exploit.should_barrel
    with fold>=0.55) misses thin-value hands. Uses 0.40-0.55x pot sizing — SMALLER
    than standard 0.60x cbet to induce calls from fold-prone rangy opponents.
    Orthogonal to passive_exploit (turn/river + passivity>=0.60) and donk_probe
    (BB donking into PFR).
    """
    if round_idx != 1 or to_call != 0:
        return {"eligible": False}
    # has_position is True when my_id == bb (we act last postflop in heads-up).
    # We want this primitive to fire when we're in position. Also fire when we
    # raised preflop and opponent checked to us (standard cbet spot), since the
    # PFR (SB) is technically out of position but the value-bet logic still applies.
    is_bb_ip = spot_info.get("has_position", False)
    last_opp = spot_info.get("last_opp_action_type")
    is_pfr_facing_check = (last_opp == "check")
    if not (is_bb_ip or is_pfr_facing_check):
        return {"eligible": False}
    if board_texture is None or board_texture.get("dynamic", True):
        return {"eligible": False}
    if board_texture.get("wetness", 1.0) > 0.30:
        return {"eligible": False}
    if made_strength < 0.42 or made_strength > 0.58:
        return {"eligible": False}
    tier = value_profile.get("tier", "none") if value_profile else "none"
    if tier in ("strong", "nut"):
        return {"eligible": False}
    confidence = opponent_model.get("confidence", 0.0)
    if confidence < 0.15:
        return {"eligible": False}
    fold_flop = opponent_model.get("fold_to_bet_flop", 0.44)
    # Fire for the 0.45-0.60 band (above 0.55 exploits.should_barrel already fires)
    if fold_flop < 0.45:
        return {"eligible": False}
    # Graduated sizing: bigger when opp folds more, capped at 0.55 (smaller than
    # standard 0.60 cbet to induce calls from fold-prone rangy opponents).
    ratio = 0.40 + 0.10 * max(0.0, fold_flop - 0.45) / 0.15  # 0.40 at 0.45 → 0.50 at 0.60
    ratio = min(ratio, 0.55)
    return {"eligible": True, "ratio": ratio, "reason": "flop_thin_value_dryboard_ip"}
```
