```python
def _count_opp_check_raises(hand_requests, my_id, opponent_id):
    """Count opponent same-street check-then-raise (CR) sequences across all
    observed hands. A CR = opponent checks on street S, then raises/allins on
    the same street S (preflop excluded). This is a trap line distinct from
    open aggression. Returns (cr_count, cr_opportunities) where opportunities
    = hands where opponent checked at least once postflop.
    """
    cr_count = 0
    cr_opportunities = 0
    for req in hand_requests:
        if opponent_can_lock_win(req, my_id):
            continue
        history = req.get("history", [])
        if not history:
            continue
        street_checked = {}      # round_idx -> True once opp checks on that street
        street_cr = {}           # round_idx -> True once opp raises after checking
        for record in history:
            if record["player_id"] != opponent_id:
                continue
            r = record["round"]
            if r == 0:
                continue        # preflop CR is a different dynamic; skip
            atype = record["action_type"]
            if atype == "check":
                street_checked[r] = True
            elif atype in ("raise", "allin") and street_checked.get(r, False):
                street_cr[r] = True
        if any(street_checked.values()):
            cr_opportunities += 1
        if any(street_cr.values()):
            cr_count += 1
    return cr_count, cr_opportunities
```

```python
import sys

def _check_raise_fold_signal(spot_info, opponent_model, made_strength, draw_strength, round_idx, value_profile):
    """Fold signal (0.0 to ~0.08) when facing a LIVE check-raise from a known
    CR trapper. Requires BOTH (a) aggregate: opponent check_raise_freq elevated
    with adequate sample confidence, and (b) spot: live same-street CR detected
    (opp checked AND bet on current street). NEVER fires for strong/nut hands
    or strong draws (value protection). The signal is SUBTRACTED from
    call_margin/jam_buffer to tighten calling vs known trappers.
    """
    if round_idx <= 0 or opponent_model is None:
        return 0.0
    cr_conf = opponent_model.get('check_raise_confidence', 0.0)
    if cr_conf < 0.25:                       # need >=~5 sample opportunities
        return 0.0
    cr_freq = opponent_model.get('check_raise_freq', 0.08)
    if cr_freq < 0.18:                        # opponent doesn't CR enough to be a trapper
        return 0.0
    live_cr = (
        spot_info.get('opp_current_round_check_count', 0) >= 1
        and spot_info.get('opp_current_round_bet_count', 0) >= 1
    )
    if not live_cr:
        return 0.0
    if value_profile is not None and value_profile.get('tier') in ('strong', 'nut'):
        return 0.0
    if draw_strength >= 0.20:                 # protect legitimate draws
        return 0.0
    if made_strength >= 0.50:                 # value-protect top-pair-good-kicker+
        return 0.0
    base = 0.03 + 0.05 * min(cr_freq - 0.18, 0.30)
    if round_idx >= 2:
        base += 0.015
    if round_idx == 3:
        base += 0.01
    base = clamp(base, 0.0, 0.08)
    if base > 0:
        sys.stderr.write(
            "[cr_fold_signal] fired cr_freq=%.2f cr_conf=%.2f cr_samp=%d "
            "street=%d made=%.2f signal=%.3f\n"
            % (cr_freq, cr_conf, opponent_model.get('check_raise_samples', 0),
               round_idx, made_strength, base)
        )
        sys.stderr.flush()
    return base
```
